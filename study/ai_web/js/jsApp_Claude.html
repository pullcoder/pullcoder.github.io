1쇼핑몰 장바구니

🎯 목표: 상품을 담고, 수량을 바꾸고, 빼고, 할인까지 적용한 결제 금액을 계산하는 장바구니 객체를 만듭니다.

객체 리터럴메서드this객체 배열속성 수정
/* ===== 실행 순서 =====
 * ① cart 객체 정의 (상품 목록 배열 + 쿠폰 + 메서드들)
 * ② add()로 상품 담기 → 같은 상품이면 수량만 증가
 * ③ changeQty()로 수량 변경, remove()로 상품 삭제
 * ④ applyCoupon()으로 할인율 설정
 * ⑤ printReceipt()로 영수증 출력
 * ====================== */

// ① 장바구니 객체
const cart = {
  items: [],          // 담긴 상품들 (객체 배열)
  discountRate: 0,    // 할인율 (0.1 = 10%)

  // ② 상품 담기
  add(name, price, qty = 1) {
    // 이미 담긴 상품인지 찾기
    const found = this.items.find(item => item.name === name);
    if (found) {
      found.qty += qty;                      // 있으면 수량만 증가
    } else {
      this.items.push({ name, price, qty }); // 없으면 새 객체 추가
    }
    console.log(`🛒 ${name} ${qty}개 담음`);
  },

  // ③-1 수량 변경
  changeQty(name, qty) {
    const found = this.items.find(item => item.name === name);
    if (!found) return console.log(`⚠️ ${name}은(는) 장바구니에 없습니다`);
    found.qty = qty;                         // 속성 수정
    console.log(`✏️ ${name} 수량 → ${qty}개`);
  },

  // ③-2 상품 삭제
  remove(name) {
    this.items = this.items.filter(item => item.name !== name);
    console.log(`🗑️ ${name} 삭제`);
  },

  // ④ 쿠폰 적용
  applyCoupon(code) {
    const coupons = { WELCOME10: 0.1, VIP20: 0.2 };  // 쿠폰표도 객체!
    if (code in coupons) {
      this.discountRate = coupons[code];             // 대괄호 접근
      console.log(`🎟️ 쿠폰 ${code} 적용 (${coupons[code] * 100}% 할인)`);
    } else {
      console.log(`❌ 존재하지 않는 쿠폰: ${code}`);
    }
  },

  // 합계 계산 (메서드 안에서 다른 메서드 호출도 this로)
  subtotal() {
    let sum = 0;
    for (const item of this.items) sum += item.price * item.qty;
    return sum;
  },

  // ⑤ 영수증 출력
  printReceipt() {
    console.log("========== 영수증 ==========");
    for (const item of this.items) {
      const line = item.price * item.qty;
      console.log(`${item.name} x${item.qty}  ${line.toLocaleString()}원`);
    }
    const sub = this.subtotal();
    const discount = Math.floor(sub * this.discountRate);
    console.log("----------------------------");
    console.log(`상품 합계 : ${sub.toLocaleString()}원`);
    console.log(`할인 금액 : -${discount.toLocaleString()}원`);
    console.log(`결제 금액 : ${(sub - discount).toLocaleString()}원`);
  }
};

// ② 담기
cart.add("무선 키보드", 45000);
cart.add("마우스 패드", 9000, 2);
cart.add("USB 허브", 23000);
cart.add("무선 키보드", 45000);   // 같은 상품 → 수량 2로

// ③ 수량 변경 / 삭제
cart.changeQty("마우스 패드", 3);
cart.remove("USB 허브");
cart.changeQty("모니터", 1);       // 없는 상품

// ④ 쿠폰
cart.applyCoupon("FREE100");       // 잘못된 쿠폰
cart.applyCoupon("WELCOME10");

// ⑤ 영수증
cart.printReceipt();
🔎 포인트쿠폰 목록 coupons처럼 "이름 → 값"을 찾는 표는 객체로 만들고 in과 대괄호 표기법으로 조회하면 if문을 여러 개 쓰지 않아도 됩니다.
🛠️ 도전 과제

  2. 결제 금액이 50,000원 이상이면 배송비 0원, 아니면 3,000원이 붙도록 영수증을 바꿔 보세요.

2학생 성적 관리

🎯 목표: 학생마다 과목별 점수 객체를 갖게 하고, for...in으로 평균을 구해 학점과 석차를 매깁니다.

객체 배열중첩 객체for...in속성 추가Object.keys
/* ===== 실행 순서 =====
 * ① 학생 목록 생성 (각 학생 안에 scores 객체가 중첩)
 * ② getGrade() : 평균 → 학점 변환 함수
 * ③ for...in으로 과목 점수 합산 → 평균·학점 속성 추가
 * ④ 평균 높은 순으로 정렬 → 석차 출력
 * ⑤ 과목별 최고 점수 학생 찾기
 * ====================== */

// ① 학생 데이터 (객체 배열 + 중첩 객체)
const students = [
  { name: "김하늘", scores: { 국어: 92, 영어: 88, 수학: 95 } },
  { name: "이바다", scores: { 국어: 78, 영어: 95, 수학: 70 } },
  { name: "박구름", scores: { 국어: 85, 영어: 80, 수학: 99 } },
  { name: "최별님", scores: { 국어: 60, 영어: 72, 수학: 65 } }
];

// ② 평균 → 학점
function getGrade(avg) {
  if (avg >= 90) return "A";
  if (avg >= 80) return "B";
  if (avg >= 70) return "C";
  if (avg >= 60) return "D";
  return "F";
}

// ③ 각 학생에게 total, avg, grade 속성을 '추가'
for (const s of students) {
  let total = 0;
  for (const subject in s.scores) {       // 국어, 영어, 수학
    total += s.scores[subject];
  }
  const count = Object.keys(s.scores).length;
  s.total = total;
  s.avg = Number((total / count).toFixed(1));  // 소수 첫째 자리
  s.grade = getGrade(s.avg);
}

// ④ 평균 내림차순 정렬 후 석차 출력
students.sort((a, b) => b.avg - a.avg);
console.log("===== 성적표 =====");
students.forEach((s, i) => {
  console.log(`${i + 1}등 ${s.name}  총점 ${s.total}  평균 ${s.avg}  학점 ${s.grade}`);
});

// ⑤ 과목별 최고점 학생 (결과도 객체에 담기)
const best = {};
for (const subject in students[0].scores) {
  let top = students[0];
  for (const s of students) {
    if (s.scores[subject] > top.scores[subject]) top = s;
  }
  best[subject] = `${top.name}(${top.scores[subject]}점)`;
}
console.log("===== 과목별 1등 =====");
console.log(best);
💡 포인트과목 이름을 코드에 직접 적지 않고 for...in으로 꺼내기 때문에, 과목이 추가돼도(예: 과학: 90) 계산 코드는 그대로 동작합니다.
🛠️ 도전 과제
  1. 모든 학생에게 과학 점수를 추가하고 다시 실행해 보세요.
  2. 학점별 인원수를 { A: 1, B: 2, ... } 형태의 객체로 세어 출력해 보세요.

3은행 계좌 시스템

🎯 목표: 클래스로 계좌 객체를 여러 개 만들고, 입금·출금·이체 시 잔액을 검사하며 거래 내역을 기록합니다.

classnewthis객체 안의 배열객체 간 상호작용
/* ===== 실행 순서 =====
 * ① Account 클래스 정의 (생성자 + 메서드)
 * ② new로 계좌 2개 생성
 * ③ 입금 / 출금 (잔액 부족 시 거절)
 * ④ transfer()로 다른 계좌 객체에 이체
 * ⑤ 각 계좌의 거래 내역 출력
 * ====================== */

// ① 계좌 설계도
class Account {
  constructor(owner, number, balance = 0) {
    this.owner = owner;       // 예금주
    this.number = number;     // 계좌번호
    this.balance = balance;   // 잔액
    this.history = [];        // 거래 내역 (객체 배열)
  }

  // 내역 기록용 (내부에서만 사용)
  log(type, amount) {
    this.history.push({ type, amount, balance: this.balance });
  }

  deposit(amount) {
    if (amount <= 0) return console.log("❌ 입금액은 0보다 커야 합니다");
    this.balance += amount;
    this.log("입금", amount);
    console.log(`💰 ${this.owner} 입금 ${amount.toLocaleString()}원`);
  }

  withdraw(amount) {
    if (amount > this.balance) {
      console.log(`❌ ${this.owner} 출금 실패 (잔액 ${this.balance.toLocaleString()}원)`);
      return false;
    }
    this.balance -= amount;
    this.log("출금", amount);
    console.log(`💸 ${this.owner} 출금 ${amount.toLocaleString()}원`);
    return true;
  }

  // ④ 다른 계좌 객체를 매개변수로 받음
  transfer(target, amount) {
    console.log(`🔁 ${this.owner} → ${target.owner} ${amount.toLocaleString()}원 이체 시도`);
    if (this.withdraw(amount)) {   // 출금 성공 시에만
      target.deposit(amount);      // 상대 계좌에 입금
    }
  }

  printHistory() {
    console.log(`----- ${this.owner} (${this.number}) 거래 내역 -----`);
    for (const h of this.history) {
      console.log(`${h.type}  ${h.amount.toLocaleString()}원  → 잔액 ${h.balance.toLocaleString()}원`);
    }
  }
}

// ② 계좌 생성
const accA = new Account("김지우", "111-222", 100000);
const accB = new Account("박서준", "333-444");

// ③ 입출금
accA.deposit(50000);
accA.withdraw(30000);
accB.withdraw(10000);        // 잔액 0 → 실패
accB.deposit(-500);          // 잘못된 금액

// ④ 이체
accA.transfer(accB, 70000);
accB.transfer(accA, 999999); // 잔액 부족 → 실패

// ⑤ 결과
accA.printHistory();
accB.printHistory();
console.log("최종 잔액:", { [accA.owner]: accA.balance, [accB.owner]: accB.balance });
🔎 포인트transfer(target, amount)처럼 객체를 다른 객체의 메서드에 넘겨 서로 상호작용하게 만드는 것이 객체 지향 프로그래밍의 핵심입니다. Java 수업의 클래스·인스턴스 개념과 똑같습니다.
💡 계산된 속성 이름마지막 줄의 { [accA.owner]: ... }처럼 키 자리에 대괄호를 쓰면 변수 값이 키 이름이 됩니다.
🛠️ 도전 과제
  1. 1회 출금 한도(예: 500,000원)를 넘으면 거절하도록 withdraw()를 수정해 보세요.
  2. 잔액에 이자율을 곱해 더해 주는 addInterest(rate) 메서드를 만들어 보세요.

4도서관 대출 관리

🎯 목표: 책 정보를 ISBN을 키로 하는 객체에 저장해 빠르게 찾고, 대출·반납·검색 기능을 만듭니다.

객체를 사전처럼대괄호 접근중첩 객체?. 옵셔널 체이닝delete
/* ===== 실행 순서 =====
 * ① library 객체 : books(ISBN → 책 정보) + 메서드
 * ② addBook()으로 책 등록 (중첩 객체 loan 포함)
 * ③ borrow() / giveBack()으로 대출·반납
 * ④ search()로 제목 키워드 검색
 * ⑤ removeBook()으로 폐기 (delete) 후 목록 출력
 * ====================== */

const library = {
  // ① 키 = ISBN, 값 = 책 객체
  books: {},

  // ② 등록
  addBook(isbn, title, author) {
    if (this.books[isbn]) return console.log(`⚠️ 이미 등록된 책: ${title}`);
    this.books[isbn] = {
      title,
      author,
      loan: null      // 대출 정보 (없으면 null)
    };
    console.log(`📚 등록: ${title}`);
  },

  // ③-1 대출
  borrow(isbn, member, days = 14) {
    const book = this.books[isbn];
    if (!book) return console.log(`❌ 없는 ISBN: ${isbn}`);
    if (book.loan) return console.log(`❌ '${book.title}'은(는) ${book.loan.member}님이 대출 중`);
    book.loan = { member, days };   // 중첩 객체로 대출 정보 저장
    console.log(`✅ ${member}님 '${book.title}' 대출 (${days}일)`);
  },

  // ③-2 반납
  giveBack(isbn) {
    const book = this.books[isbn];
    // ?. : book이 없거나 loan이 null이어도 에러 없이 undefined
    const who = book?.loan?.member;
    if (!who) return console.log(`⚠️ 반납할 대출 기록 없음 (${isbn})`);
    book.loan = null;
    console.log(`↩️ ${who}님 '${book.title}' 반납 완료`);
  },

  // ④ 검색 : for...in으로 모든 책 확인
  search(keyword) {
    const result = [];
    for (const isbn in this.books) {
      const b = this.books[isbn];
      if (b.title.includes(keyword)) {
        result.push(`${b.title} / ${b.author} / ${b.loan ? "대출중" : "대출가능"}`);
      }
    }
    console.log(`🔍 '${keyword}' 검색 결과 ${result.length}건`, result);
  },

  // ⑤ 폐기
  removeBook(isbn) {
    const title = this.books[isbn]?.title;
    delete this.books[isbn];        // 속성 삭제
    console.log(`🗑️ 폐기: ${title}`);
  },

  printAll() {
    console.log("===== 소장 도서 =====");
    for (const isbn in this.books) {
      const b = this.books[isbn];
      const state = b.loan ? `대출중(${b.loan.member})` : "대출가능";
      console.log(`[${isbn}] ${b.title} - ${state}`);
    }
  }
};

// ② 등록
library.addBook("978-01", "자바스크립트 입문", "홍길동");
library.addBook("978-02", "자바 객체지향 프로그래밍", "이순신");
library.addBook("978-03", "모던 자바스크립트 실전", "강감찬");
library.addBook("978-01", "자바스크립트 입문", "홍길동");  // 중복

// ③ 대출 / 반납
library.borrow("978-01", "jw");
library.borrow("978-01", "민지");      // 이미 대출 중
library.borrow("978-99", "민지");      // 없는 책
library.borrow("978-03", "민지", 7);
library.giveBack("978-01");
library.giveBack("978-02");            // 대출 기록 없음

// ④ 검색
library.search("자바스크립트");

// ⑤ 폐기 후 전체 목록
library.removeBook("978-02");
library.printAll();
💡 왜 배열이 아니라 객체에 저장할까?배열이면 책 하나를 찾을 때마다 처음부터 끝까지 훑어야 하지만, ISBN을 키로 쓰면 books[isbn] 한 번으로 바로 찾습니다. 고유한 ID로 찾는 데이터는 객체가 유리합니다.
🛠️ 도전 과제
  1. 특정 회원이 빌린 책 목록을 출력하는 listByMember(member)를 만들어 보세요.
  2. 한 회원이 최대 2권까지만 빌릴 수 있게 borrow()를 수정해 보세요.

5단어 빈도 분석기

🎯 목표: 문장을 단어로 나누고, 단어마다 몇 번 나왔는지 객체로 세어 많이 나온 순으로 그래프처럼 출력합니다. (리뷰 분석, 검색어 통계 등에 그대로 쓰이는 패턴)

동적 속성 추가대괄호 접근Object.entries객체 → 배열 변환
/* ===== 실행 순서 =====
 * ① 분석할 리뷰 문장 준비
 * ② 소문자 변환 · 문장부호 제거 · 공백으로 단어 분리
 * ③ 빈 객체 counter에 단어를 키로 개수 누적
 * ④ Object.entries로 배열 변환 → 많은 순 정렬
 * ⑤ 상위 5개를 막대그래프로 출력
 * ====================== */

// ① 리뷰 데이터
const reviews = `배송이 빠르고 포장이 좋아요. 가격도 좋아요!
품질이 좋아요 배송도 빠르고 만족합니다.
가격이 조금 비싸지만 품질은 최고. 배송 빠르고 좋아요`;

// ② 단어 분리 (정규식: 한글/영문/숫자/공백만 남김)
const words = reviews
  .toLowerCase()
  .replace(/[^가-힣a-z0-9\s]/g, "")
  .split(/\s+/)
  .filter(w => w.length > 0);
console.log("전체 단어 수:", words.length);

// ③ 개수 세기 : 키가 없으면 0에서 시작
const counter = {};
for (const w of words) {
  counter[w] = (counter[w] || 0) + 1;   // 없는 속성 → undefined → 0 처리
}
console.log("종류 수:", Object.keys(counter).length);

// ④ [단어, 개수] 배열로 바꿔서 정렬
const ranking = Object.entries(counter)
  .sort((a, b) => b[1] - a[1]);

// ⑤ 상위 5개 막대그래프
console.log("===== 많이 나온 단어 TOP 5 =====");
ranking.slice(0, 5).forEach(([word, n], i) => {
  const bar = "■".repeat(n);
  console.log(`${i + 1}. ${word.padEnd(4, "　")} ${bar} ${n}`);
});

// 보너스 : 특정 단어 조회
const target = "가격도";
console.log(`'${target}' 등장 횟수:`, counter[target] ?? 0);
🔎 핵심 한 줄counter[w] = (counter[w] || 0) + 1 — 처음 보는 단어면 속성이 없어서 undefined가 되고, || 0 덕분에 0부터 셉니다. 객체에 속성을 동적으로 추가하는 가장 흔한 실전 패턴입니다.
