Intro.java

package chap01;

public class Intro {

	
	public static void main(String[] args) {
		
		
		String st1 = "안녕하세요?";
		String st2 = "저는 자바를 처음 공부합니다.";
		String st3 = "저의 나이는 20살입니다.";
				
		System.out.println(st1+"\n"+st2+"\n"+st3);
		
	}

}
